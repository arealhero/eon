     1 |           ASSIGN           VARIABLE s32 a@1, CONSTANT s32 10
     2 |           ASSIGN           VARIABLE s32 a@2, CONSTANT s32 20
     3 |           RETURN

parameter_reassignment:
     1 |           GET_PARAMETER    VARIABLE s32 parameter@1, ARGUMENT 0
     2 |           ASSIGN           VARIABLE s32 parameter@2, CONSTANT s32 20
     3 |           RETURN

simple_conditional_assignment:
     1 |           ASSIGN           VARIABLE s32 a@1, CONSTANT s32 10
     2 |           JUMP             LABEL_1
     3 | LABEL_1:

       |
       |           PHI              VARIABLE s32 a@3, VARIABLE s32 a@1
     5 | LABEL_2:
     6 |           RETURN           VARIABLE s32 a@3
